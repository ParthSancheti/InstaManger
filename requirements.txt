# =============================================================
#  instagrapi — DIRECT upload fallback for Reels and Stories.
#  Graph downloads media from a public URL; ImgBB is images-only

#  Optional: without it, video publishing depends on hosting.
# =============================================================
instagrapi

# =============================================================
#  imageio-ffmpeg — bundles an ffmpeg binary. REQUIRED unless you

#  publishing — without it Meta reports "processing ERROR".
# =============================================================
imageio-ffmpeg
